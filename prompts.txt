### 1. Enredo e Ambientação (Narrativa Interativa)
- **Tema:** O Arrebatamento cristão aconteceu e o protagonista (jogador) ficou para trás.
- **Aliados:** O jogador conta com o apoio de um pastor experiente na fé e de uma amiga de longa data que também ficaram.
- **Antagonistas e Personagens Bíblicos:** Ao longo da jornada, o jogador enfrentará escolhas morais, espirituais e de sobrevivência diante da ascensão do Anticristo, a influência enganosa do Falso Profeta e o testemunho público das Duas Testemunhas.
- **Clímax:** A narrativa deve culminar na Grande Tribulação e na Gloriosa Volta de Jesus Cristo.

### 2. Mecânica de Jogo e Múltiplos Finais
- **Sistema de Escolhas:** Cada cena/pergunta apresenta uma situação crítica com 2 a 4 opções de múltipla escolha.
- **Atributos do Personagem (Opcional/Dinâmico):** O jogador deve gerenciar medidores de "Fé/Esperança" e "Sobrevivência" que mudam de acordo com as decisões tomadas.
- **Múltiplos Finais:** O jogo deve possuir **pelo menos 4 finais diferentes**, dependendo das escolhas de fidelidade à fé, apostasia, coragem ou acovardamento perante o sistema do Anticristo:
  1. *Final da Coroa da Vida / Fidelidade Suprema:* Mantém a fé até o fim, resiste ao sistema e aguarda a Volta de Jesus (Final Positivo Principal).
  2. *Final da Marca / Apostasia:* Cede à pressão do sistema, ganha conforto temporário, mas perde a salvação espiritual.
  3. *Final do Mártir:* É pego pregando ao lado das Duas Testemunhas, selando seu testemunho com a vida.
  4. *Final da Fuga / Ocultação:* Sobrevive escondido nas sombras, mas com forte angústia e fé enfraquecida até o fim dos tempos.

### 3. Requisitos de Design e Interface (UI/UX)
- **Estética Visual:** Interface imersiva, com tons escuros (preto/cinza chumbo), detalhes em dourado celestial e vermelho/roxo sombrio para momentos de tensão com o Anticristo.
- **Responsividade:** Layout adaptado para funcionar perfeitamente em computadores e celulares.
- **Feedback Visual:** Animações suaves de transição de texto (efeito máquina de escrever ou fade-in) e barras visuais atualizadas para os status de Fé e Vida.

function updateMeters() {
  els.faithValue.textContent = state.faith;

  els.survivalBar.style.width = `${state.survival}%`;

function applyEffects(effects = {}) {
  state.faith = clamp(state.faith + (effects.faith || 0));
  state.survival = clamp(state.survival + (effects.survival || 0));
  state.witness += effects.witness || 0;
  updateMeters();
}

function showConsequence(message, effects = {}) {
  const parts = [message];
  if (effects.faith) parts.push(`Fé: ${effects.faith > 0 ? "+" : ""}${effects.faith}`);
  if (effects.survival) parts.push(`Sobrevivência: ${effects.survival > 0 ? "+" : ""}${effects.survival}`);
  if (effects.witness) parts.push(`Testemunho: +${effects.witness}`);
  const tone = (effects.faith < 0 || effects.survival < 0) ? "negative" :
    (effects.faith > 0 || effects.survival > 0 || effects.witness > 0) ? "positive" : "neutral";
  els.consequence.textContent = parts.join(" • ");
  els.consequence.className = `consequence ${tone}`;
}

function typeText(target, text, speed = TYPE_SPEED) {
  clearInterval(typingTimer);
  target.textContent = "";
  let index = 0;
  typingTimer = setInterval(() => {
    target.textContent += text[index++];
    if (index >= text.length) {
      clearInterval(typingTimer);
      typingTimer = null;
    }
  }, speed);
}

const clamp = (value, min = 0, max = 100) =>
  Math.max(min, Math.min(max, value));

let typingTimer = null;
let transitionTimer = null;
let inputLocked = false;

function showScreen(name) {
  Object.values(screens).forEach(screen =>
    screen?.classList.remove("active")
  );

  if (!screens[name]) {
    console.error(`Tela não encontrada: ${name}`);
    return;
  }

  screens[name].classList.add("active");
}

function updateMeters() {
  els.faithValue.textContent = state.faith;
  els.survivalValue.textContent = state.survival;

  els.faithBar.style.width = `${state.faith}%`;
  els.survivalBar.style.width = `${state.survival}%`;
}

function applyEffects(effects = {}) {
  state.faith = clamp(state.faith + (effects.faith || 0));
  state.survival = clamp(state.survival + (effects.survival || 0));
  state.witness += effects.witness || 0;

  updateMeters();
}

function showConsequence(message, effects = {}) {
  const parts = [message];

  if (effects.faith) {
    parts.push(`Fé: ${effects.faith > 0 ? "+" : ""}${effects.faith}`);
  }

  if (effects.survival) {
    parts.push(`Sobrevivência: ${effects.survival > 0 ? "+" : ""}${effects.survival}`);
  }

  if (effects.witness) {
    parts.push(`Testemunho: +${effects.witness}`);
  }

  const tone =
    effects.faith < 0 || effects.survival < 0
      ? "negative"
      : effects.faith > 0 || effects.survival > 0 || effects.witness > 0
        ? "positive"
        : "neutral";

  els.consequence.textContent = parts.join(" • ");
  els.consequence.className = `consequence ${tone}`;
}

function typeText(target, text, speed = TYPE_SPEED) {
  clearInterval(typingTimer);
  target.textContent = "";

  let index = 0;

  typingTimer = setInterval(() => {
    target.textContent += text[index++];

    if (index >= text.length) {
      clearInterval(typingTimer);
      typingTimer = null;
    }
  }, speed);
}

function renderNode(nodeId) {
  clearTimeout(transitionTimer);
  clearInterval(typingTimer);

  inputLocked = false;

  const node = story[nodeId];

  if (!node) {
    console.error(`Cena não encontrada: ${nodeId}`);
    return;
  }

  if (node.ending) {
    resolveEnding(node.ending);
    return;
  }

  state.currentNode = nodeId;

  els.chapter.textContent = node.chapter || "";
  els.location.textContent = node.location || "";
  els.sceneTag.textContent = node.tag || "Cena";
  els.sceneTitle.textContent = node.title || "";

  els.consequence.className = "consequence hidden";
  els.consequence.textContent = "";

  els.choices.innerHTML = "";

  typeText(els.storyText, node.text || "");

  (node.choices || []).forEach(option => {
    const button = document.createElement("button");

    button.type = "button";
    button.className =
      `choice ${
        option.effects?.faith < 0 || option.effects?.survival < 0
          ? "danger"
          : ""
      }`;

    button.textContent = option.label;

    button.addEventListener("click", () => chooseOption(option));

    els.choices.appendChild(button);
  });
}

function chooseOption(option) {
  if (inputLocked) return;

  inputLocked = true;

  [...els.choices.querySelectorAll("button")].forEach(button => {
    button.disabled = true;
  });

  applyEffects(option.effects);

  if (typeof option.action === "function") {
    option.action(state);
  }

  showConsequence(option.consequence || "Sua escolha foi registrada.", option.effects);

  transitionTimer = setTimeout(() => {
    renderNode(option.next);
  }, CONSEQUENCE_DELAY);
}

const clamp = (value, min = 0, max = 100) =>
  Math.max(min, Math.min(max, value));

let typingTimer = null;
let transitionTimer = null;
let inputLocked = false;

function showScreen(name) {
  Object.values(screens).forEach(screen =>
    screen?.classList.remove("active")
  );

  if (!screens[name]) {
    console.error(`Tela não encontrada: ${name}`);
    return;
  }

  screens[name].classList.add("active");
}

function updateMeters() {
  els.faithValue.textContent = state.faith;
  els.survivalValue.textContent = state.survival;

  els.faithBar.style.width = `${state.faith}%`;
  els.survivalBar.style.width = `${state.survival}%`;
}

function applyEffects(effects = {}) {
  state.faith = clamp(state.faith + (effects.faith || 0));
  state.survival = clamp(state.survival + (effects.survival || 0));
  state.witness += effects.witness || 0;

  updateMeters();
}

function showConsequence(message, effects = {}) {
  const parts = [message];

  if (effects.faith) {
    parts.push(`Fé: ${effects.faith > 0 ? "+" : ""}${effects.faith}`);
  }

  if (effects.survival) {
    parts.push(`Sobrevivência: ${effects.survival > 0 ? "+" : ""}${effects.survival}`);
  }

  if (effects.witness) {
    parts.push(`Testemunho: +${effects.witness}`);
  }

  const tone =
    effects.faith < 0 || effects.survival < 0
      ? "negative"
      : effects.faith > 0 || effects.survival > 0 || effects.witness > 0
        ? "positive"
        : "neutral";

  els.consequence.textContent = parts.join(" • ");
  els.consequence.className = `consequence ${tone}`;
}

function typeText(target, text, speed = TYPE_SPEED) {
    // Interrompe uma digitação anterior, caso exista.
    if (typingInterval !== null) {
        clearInterval(typingInterval);
        typingInterval = null;
    }

    typingTarget = target;
    typingFullText = text;
    typingIndex = 0;

    target.textContent = "";

    if (skipDialogueButton) {
        skipDialogueButton.hidden = false;
    }

    typingInterval = setInterval(() => {
        if (typingIndex < typingFullText.length) {
            target.textContent += typingFullText.charAt(typingIndex);
            typingIndex++;
        } else {
            finishTyping();
        }
    }, speed);
}

function renderNode(nodeId) {
  clearTimeout(transitionTimer);
  clearInterval(typingTimer);

  inputLocked = false;

  const node = story[nodeId];

  if (!node) {
    console.error(`Cena não encontrada: ${nodeId}`);
    return;
  }

  if (node.ending) {
    resolveEnding(node.ending);
    return;
  }

  state.currentNode = nodeId;

  els.chapter.textContent = node.chapter || "";
  els.location.textContent = node.location || "";
  els.sceneTag.textContent = node.tag || "Cena";
  els.sceneTitle.textContent = node.title || "";

  els.consequence.className = "consequence hidden";
  els.consequence.textContent = "";

  els.choices.innerHTML = "";

  typeText(els.storyText, node.text || "");

  (node.choices || []).forEach(option => {
    const button = document.createElement("button");

    button.type = "button";
    button.className =
      `choice ${
        option.effects?.faith < 0 || option.effects?.survival < 0
          ? "danger"
          : ""
      }`;

    button.textContent = option.label;

    button.addEventListener("click", () => chooseOption(option));

    els.choices.appendChild(button);
  });
}

function chooseOption(option) {
  if (inputLocked) return;

  inputLocked = true;

  [...els.choices.querySelectorAll("button")].forEach(button => {
    button.disabled = true;
  });

  applyEffects(option.effects);

  if (typeof option.action === "function") {
    option.action(state);
  }

  showConsequence(option.consequence || "Sua escolha foi registrada.", option.effects);

  transitionTimer = setTimeout(() => {
    renderNode(option.next);
  }, CONSEQUENCE_DELAY);
}

function resolveEnding(endingId) {
  const ending = endings[endingId];

  if (!ending) {
    console.error(`Final não encontrado: ${endingId}`);
    return;
  }

  clearInterval(typingTimer);
  clearTimeout(transitionTimer);

  finaisConcluidos.add(endingId);
  salvarProgresso();

  els.endingEyebrow.textContent = ending.eyebrow;
  els.endingTitle.textContent = ending.title;

  typeText(els.endingText, ending.text, 9);

  els.endingFaith.textContent = state.faith;
  els.endingSurvival.textContent = state.survival;
  els.endingWitness.textContent = state.witness;

  showScreen("ending");

  if (ENDING_IDS.every(id => finaisConcluidos.has(id))) {

/* ---------- PROGRESSO DOS OITO FINAIS ---------- */
const STORAGE_KEY = "finaisConcluidos";
const ENDING_IDS = [
  "cadeia", "lago_fogo", "coroa_vida", "marca",
  "solidao", "martir", "dependente", "a_fuga"
];

function carregarFinaisConcluidos() {
  try {
    const dados = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
    return new Set(
      Array.isArray(dados)
        ? dados.filter(id => ENDING_IDS.includes(id))
        : []
    );
  } catch (error) {
    console.warn("Não foi possível carregar o progresso:", error);
    return new Set();
  }
}

const finaisConcluidos = carregarFinaisConcluidos();

const clamp = (value, min = 0, max = 100) =>
  Math.max(min, Math.min(max, value));

let typingTimer = null;
let transitionTimer = null;
let inputLocked = false;

function showScreen(name) {
  Object.values(screens).forEach(screen =>
    screen?.classList.remove("active")
  );

  if (!screens[name]) {
    console.error(`Tela não encontrada: ${name}`);
    return;
  }

  screens[name].classList.add("active");
}

function updateMeters() {
  els.faithValue.textContent = state.faith;
  els.survivalValue.textContent = state.survival;

  els.faithBar.style.width = `${state.faith}%`;
  els.survivalBar.style.width = `${state.survival}%`;
}

function applyEffects(effects = {}) {
  state.faith = clamp(state.faith + (effects.faith || 0));
  state.survival = clamp(state.survival + (effects.survival || 0));
  state.witness += effects.witness || 0;

  updateMeters();
}

function showConsequence(message, effects = {}) {
  const parts = [message];

  if (effects.faith) {
    parts.push(`Fé: ${effects.faith > 0 ? "+" : ""}${effects.faith}`);
  }

  if (effects.survival) {
    parts.push(`Sobrevivência: ${effects.survival > 0 ? "+" : ""}${effects.survival}`);
  }

  if (effects.witness) {
    parts.push(`Testemunho: +${effects.witness}`);
  }

  const tone =
    effects.faith < 0 || effects.survival < 0
      ? "negative"
      : effects.faith > 0 || effects.survival > 0 || effects.witness > 0
        ? "positive"
        : "neutral";

  els.consequence.textContent = parts.join(" • ");
  els.consequence.className = `consequence ${tone}`;
}
